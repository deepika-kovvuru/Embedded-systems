
EXP-06    ARDUINO TO 16x2 LCD DISPLAY USING PROTEUS 

SOURCE CODE:
#include <LiquidCrystal.h>

LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

void setup() 
{
  lcd.begin(16, 2);
}

void loop() 
{
  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("Hello");
  delay(1000);
  lcd.setCursor(0,1);
  lcd.print("CS01 here!");
  delay(1000);
}
